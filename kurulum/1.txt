create schema if not exists extensions;
create extension if not exists btree_gist with schema extensions;
create table if not exists public.shops (
id              uuid primary key default gen_random_uuid(),
owner_id        uuid not null unique references auth.users(id) on delete cascade,
name            text not null check (char_length(name) between 2 and 80),
slug            text not null unique check (slug ~ '^[a-z0-9-]{3,70}$'),
city            text not null default 'Ankara',
district        text not null default '',
address         text not null default '',
phone           text not null default '',
instagram       text not null default '',
lat             double precision check (lat between -90 and 90),
lng             double precision check (lng between -180 and 180),
photos          text[] not null default '{}',
accepting       boolean not null default true,
is_active       boolean not null default true,
auto_pricing    boolean not null default true,
surge_threshold int not null default 15 check (surge_threshold between 1 and 200),
surge_percent   int not null default 15 check (surge_percent between 0 and 50),
slot_minutes    int not null default 30 check (slot_minutes in (15, 30, 60)),
hours           jsonb not null default '{"0":null,"1":["10:00","20:00"],"2":["10:00","20:00"],"3":["10:00","20:00"],"4":["10:00","20:00"],"5":["10:00","20:00"],"6":["10:00","20:00"]}'
check (jsonb_typeof(hours) = 'object'),
created_at      timestamptz not null default now()
);
create table if not exists public.staff (
id          uuid primary key default gen_random_uuid(),
shop_id     uuid not null references public.shops(id) on delete cascade,
name        text not null check (char_length(name) between 2 and 60),
title       text not null default 'Usta' check (char_length(title) <= 30),
commission  int not null default 40 check (commission between 0 and 100),
active      boolean not null default true,
created_at  timestamptz not null default now()
);
create index if not exists staff_shop_idx on public.staff(shop_id);
create table if not exists public.services (
id        uuid primary key default gen_random_uuid(),
staff_id  uuid not null references public.staff(id) on delete cascade,
shop_id   uuid not null references public.shops(id) on delete cascade,
name      text not null check (char_length(name) between 2 and 40),
price     int not null check (price between 0 and 100000),
duration  int not null check (duration between 5 and 480),
active    boolean not null default true,
unique (staff_id, name)
);
create index if not exists services_shop_idx on public.services(shop_id);
create table if not exists public.appointments (
id              uuid primary key default gen_random_uuid(),
shop_id         uuid not null references public.shops(id) on delete cascade,
staff_id        uuid not null references public.staff(id) on delete cascade,
service_id      uuid references public.services(id) on delete set null,
service_name    text not null,
customer_name   text not null check (char_length(customer_name) between 1 and 60),
customer_phone  text not null default '',
starts_at       timestamptz not null,
ends_at         timestamptz not null,
base_price      int not null default 0,
price           int not null default 0,
surge           boolean not null default false,
status          text not null default 'confirmed' check (status in ('confirmed','completed','cancelled','no_show')),
source          text not null default 'web' check (source in ('web','barber','walkin')),
token           uuid not null unique default gen_random_uuid(),
created_at      timestamptz not null default now(),
check (ends_at > starts_at),
constraint appointments_no_overlap exclude using gist
(staff_id with =, tstzrange(starts_at, ends_at, '[)') with &&)
where (status in ('confirmed','completed') and source <> 'walkin')
);
create index if not exists appointments_shop_time_idx on public.appointments(shop_id, starts_at);
create table if not exists public.holds (
id          uuid primary key default gen_random_uuid(),
shop_id     uuid not null references public.shops(id) on delete cascade,
staff_id    uuid not null references public.staff(id) on delete cascade,
service_id  uuid not null references public.services(id) on delete cascade,
starts_at   timestamptz not null,
ends_at     timestamptz not null,
base_price  int not null,
price       int not null,
surge       boolean not null,
expires_at  timestamptz not null,
constraint holds_no_overlap exclude using gist
(staff_id with =, tstzrange(starts_at, ends_at, '[)') with &&)
);
create table if not exists public.reviews (
id              uuid primary key default gen_random_uuid(),
shop_id         uuid not null references public.shops(id) on delete cascade,
appointment_id  uuid unique references public.appointments(id) on delete set null,
author          text not null,
rating          int not null check (rating between 1 and 5),
body            text not null check (char_length(body) between 3 and 500),
reply           text check (reply is null or char_length(reply) <= 500),
replied_at      timestamptz,
created_at      timestamptz not null default now()
);
create index if not exists reviews_shop_idx on public.reviews(shop_id);
create table if not exists public.waitlist (
id          uuid primary key default gen_random_uuid(),
shop_id     uuid not null references public.shops(id) on delete cascade,
staff_id    uuid references public.staff(id) on delete cascade,
service_id  uuid references public.services(id) on delete set null,
day         date not null,
name        text not null,
phone       text not null,
status      text not null default 'waiting' check (status in ('waiting','notified')),
created_at  timestamptz not null default now(),
unique (shop_id, day, phone)
);
create table if not exists public.walkins (
id          uuid primary key default gen_random_uuid(),
shop_id     uuid not null references public.shops(id) on delete cascade,
staff_id    uuid not null references public.staff(id) on delete cascade,
service_id  uuid not null references public.services(id) on delete cascade,
name        text not null check (char_length(name) between 1 and 60),
phone       text not null default '',
created_at  timestamptz not null default now()
);
create table if not exists public.customer_notes (
shop_id     uuid not null references public.shops(id) on delete cascade,
ckey        text not null,
note        text not null default '' check (char_length(note) <= 300),
updated_at  timestamptz not null default now(),
primary key (shop_id, ckey)
);
