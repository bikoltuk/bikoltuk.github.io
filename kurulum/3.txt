alter table public.staff          enable row level security;
alter table public.services       enable row level security;

drop policy if exists shops_read on public.shops;
create policy shops_read on public.shops for select using (is_active or owner_id = auth.uid());
drop policy if exists shops_insert on public.shops;
create policy shops_insert on public.shops for insert to authenticated with check (owner_id = auth.uid());
drop policy if exists shops_update on public.shops;
create policy shops_update on public.shops for update to authenticated using (owner_id = auth.uid()) with check (owner_id = auth.uid());
drop policy if exists staff_read on public.staff;
create policy staff_read on public.staff for select
using (exists (select 1 from shops s where s.id = shop_id and (s.is_active or s.owner_id = auth.uid())));
drop policy if exists staff_write on public.staff;
create policy staff_write on public.staff for all to authenticated
using (public.is_my_shop(shop_id)) with check (public.is_my_shop(shop_id));
drop policy if exists services_read on public.services;
create policy services_read on public.services for select
using (exists (select 1 from shops s where s.id = shop_id and (s.is_active or s.owner_id = auth.uid())));
drop policy if exists services_write on public.services;
create policy services_write on public.services for all to authenticated
using (public.is_my_shop(shop_id)) with check (public.is_my_shop(shop_id));
drop policy if exists appointments_owner on public.appointments;
create policy appointments_owner on public.appointments for all to authenticated
using (public.is_my_shop(shop_id)) with check (public.is_my_shop(shop_id));
drop policy if exists reviews_read on public.reviews;
create policy reviews_read on public.reviews for select using (true);
drop policy if exists reviews_reply on public.reviews;
create policy reviews_reply on public.reviews for update to authenticated
using (public.is_my_shop(shop_id)) with check (public.is_my_shop(shop_id));
drop policy if exists waitlist_owner on public.waitlist;
create policy waitlist_owner on public.waitlist for all to authenticated
using (public.is_my_shop(shop_id)) with check (public.is_my_shop(shop_id));
drop policy if exists walkins_owner on public.walkins;
create policy walkins_owner on public.walkins for all to authenticated
using (public.is_my_shop(shop_id)) with check (public.is_my_shop(shop_id));
drop policy if exists notes_owner on public.customer_notes;
create policy notes_owner on public.customer_notes for all to authenticated
using (public.is_my_shop(shop_id)) with check (public.is_my_shop(shop_id));
