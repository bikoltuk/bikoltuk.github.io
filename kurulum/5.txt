create or replace function public.my_staff_ids() returns setof uuid
language sql stable security definer set search_path = public as $$
select sa.staff_id from staff_accounts sa join staff st on st.id = sa.staff_id
where sa.user_id = auth.uid() and st.active
$$;
create or replace function public.is_staff_member(p_shop uuid) returns boolean
language sql stable security definer set search_path = public as $$
select exists (select 1 from staff_accounts sa join staff st on st.id = sa.staff_id
where sa.shop_id = p_shop and sa.user_id = auth.uid() and st.active)
$$;
create or replace function public.staff_invite(p_staff uuid) returns text
language plpgsql volatile security definer set search_path = public as $$
declare sh uuid; c text;
begin
select shop_id into sh from staff where id = p_staff;
if sh is null or not public.is_my_shop(sh) then raise exception 'NOT_ALLOWED'; end if;
c := upper(substr(md5(random()::text || clock_timestamp()::text || p_staff::text), 1, 8));
insert into staff_accounts (staff_id, shop_id, code, code_expires) values (p_staff, sh, c, now() + interval '7 days')
on conflict (staff_id) do update set code = excluded.code, code_expires = excluded.code_expires;
return c;
end $$;
create or replace function public.claim_staff(p_code text) returns json
language plpgsql volatile security definer set search_path = public as $$
declare r staff_accounts; s_name text; st_name text;
begin
if auth.uid() is null then raise exception 'NOT_AUTHENTICATED'; end if;
if exists (select 1 from shops where owner_id = auth.uid()) then raise exception 'OWNS_SHOP'; end if;
select * into r from staff_accounts where code = upper(btrim(coalesce(p_code, ''))) and code_expires > now() for update;
if not found then raise exception 'INVITE_INVALID'; end if;
delete from staff_accounts where user_id = auth.uid() and staff_id <> r.staff_id;
update staff_accounts set user_id = auth.uid(), code = null, code_expires = null where staff_id = r.staff_id;
select s.name, st.name into s_name, st_name from staff st join shops s on s.id = st.shop_id where st.id = r.staff_id;
return json_build_object('shop', s_name, 'staff', st_name);
end $$;

create or replace function public.confirm_booking(p_hold uuid, p_name text, p_phone text)
returns json language plpgsql volatile security definer set search_path = public as $$
declare
h holds; v services; ap appointments; lim int;
nm text := btrim(coalesce(p_name, ''));
ph text := public.norm_phone(p_phone);
begin
if char_length(nm) < 2 or char_length(nm) > 60 then raise exception 'NAME_INVALID'; end if;
if ph is null then raise exception 'PHONE_INVALID'; end if;
select * into h from holds where id = p_hold and expires_at > now() for update;
if not found then raise exception 'HOLD_EXPIRED'; end if;
if (select count(*) from appointments
where shop_id = h.shop_id and customer_phone = ph and status = 'confirmed' and starts_at > now()) >= 3 then
raise exception 'TOO_MANY';
end if;
select noshow_limit into lim from shops where id = h.shop_id;
if coalesce(lim, 0) > 0 and (select count(*) from appointments
where shop_id = h.shop_id and customer_phone = ph and status = 'no_show' and starts_at > now() - interval '180 days') >= lim then
raise exception 'NOSHOW_BLOCKED';
end if;
select * into v from services where id = h.service_id;
delete from holds where id = h.id;
begin
insert into appointments (shop_id, staff_id, service_id, service_name, customer_name, customer_phone,
starts_at, ends_at, base_price, price, surge, status, source)
values (h.shop_id, h.staff_id, h.service_id, v.name, nm, ph,
h.starts_at, h.ends_at, h.base_price, h.price, h.surge, 'confirmed', 'web')
returning * into ap;
exception when exclusion_violation then
raise exception 'SLOT_TAKEN';
end;
return json_build_object('token', ap.token, 'starts_at', ap.starts_at, 'price', ap.price, 'surge', ap.surge);
end $$;
create or replace function public.my_extras(p_tokens uuid[])
returns table (token uuid, shop_slug text, staff_id uuid, service_id uuid, loyalty_target int, loyalty_reward text,
visits int, google_review_url text)
language sql stable security definer set search_path = public as $$
select a.token, s.slug, a.staff_id, a.service_id, s.loyalty_target, s.loyalty_reward,
(select count(*) from appointments x where x.shop_id = a.shop_id and a.customer_phone <> ''
and x.customer_phone = a.customer_phone and x.status = 'completed' and x.kind = 'appointment')::int,
s.google_review_url
from appointments a join shops s on s.id = a.shop_id
where a.token = any (p_tokens[1:50])
$$;
